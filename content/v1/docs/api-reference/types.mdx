`logLevel` option is accepted by `DatabaseConfig` but is currently reserved
and does not change library logging.

## ID

```ts
type ID = number | string
```

## FieldType

```ts
type FieldType = "string" | "number" | "boolean" | "object" | "array"
```

## FieldDefinition

```ts
interface FieldDefinition {
  type: FieldType
  required?: boolean
  default?: unknown
  validate?: "email" | "url" | RegExp | ((value: unknown) => boolean)
  min?: number
  max?: number
  maxLength?: number
  items?: FieldDefinition
  unique?: boolean
}
```

```ts
interface IndexDefinition {
```

## QueryOperator

```ts
type QueryOperator = "==" | "!=" | ">" | "<" | ">=" | "<="
```

## QueryCondition

```ts
interface QueryCondition {
  type: "where" | "search" | "match"
```

## QueryStrategy

```ts
type QueryStrategy = "index_scan" | "full_scan" | "id_lookup"
```

## QueryRange

```ts
type QueryKey = string | number | Date

interface QueryRange {
  readonly lower?: QueryKey
  readonly upper?: QueryKey
  readonly lowerOpen: boolean
  readonly upperOpen: boolean
}
```

This storage-neutral range descriptor lets adapters implement indexed
lookups without depending on the browser's `IDBKeyRange` API.

## CtroDBPlugin

```ts
interface CtroDBPlugin {
  name: string
  version?: string
  storeNames?: (string | PluginStoreConfig)[]
  onDatabaseInit?(db: unknown): void | Promise<void>
  onDatabaseDestroy?(db: unknown): void | Promise<void>
  onCollectionInit?(collection: unknown): void
  onBeforeCreate?(collection: string, data: unknown, context?: WriteContext): unknown
  onAfterCreate?(collection: string, record: unknown, context?: WriteContext): void
  onAfterCreateMany?(collection: string, records: unknown[], context?: WriteContext): void
  onBeforeUpdate?(collection: string, id: ID, changes: unknown, context?: WriteContext): unknown
  onAfterUpdate?(collection: string, id: ID, record: unknown, oldRecord?: unknown, context?: WriteContext): void
  onBeforeDelete?(collection: string, id: ID, context?: WriteContext): void
  onAfterDelete?(collection: string, id: ID, oldRecord?: unknown, context?: WriteContext): void
}
```

`onDatabaseDestroy` is called when the database disconnects. Hook callbacks may
return promises; ctrodb awaits them.

## WriteContext

```ts
type WriteOrigin = "local" | "remote"

interface WriteContext {
  readonly origin: WriteOrigin
}
```

Write hooks receive the context as their final argument. Sync-applied writes
use `"remote"`; ordinary collection writes use `"local"`.

## TransactionContext

```ts
interface TransactionContext {
  collection(name: string): {
    create(data: Record<string, unknown>): Promise<Record<string, unknown>>
    findById(id: ID): Promise<Record<string, unknown> | undefined>
    findAll(): Promise<Record<string, unknown>[]>
    update(id: ID, changes: Record<string, unknown>): Promise<Record<string, unknown>>
    delete(id: ID): Promise<void>
  }
}
```

## StorageAdapter

```ts
interface StorageAdapter {
  readonly name: string
  connect(name: string, schema: SchemaConfig | null): Promise<void>
  disconnect(): Promise<void>
  isConnected(): boolean
  setConnectionChangeHandler?(handler: (event: ConnectionChangeEvent) => void): void
  getSchemaVersion(): Promise<number>
  setSchemaVersion(version: number): Promise<void>
  create(collection: string, data: unknown): Promise<unknown>
  createMany?(collection: string, records: unknown[]): Promise<unknown[]>
  findById(collection: string, id: ID): Promise<unknown>
  findByIds?(collection: string, ids: ID[]): Promise<(unknown | undefined)[]>
  findAll(collection: string): Promise<unknown[]>
  count?(collection: string): Promise<number>
  hasIndex?(collection: string, indexName: string): boolean
  update(collection: string, id: ID, changes: unknown): Promise<unknown>
  delete(collection: string, id: ID): Promise<void>
  deleteMany(collection: string, ids: ID[]): Promise<void>
  scanIndex(collection: string, indexName: string, range: QueryRange | undefined, postFilters: QueryCondition[]): Promise<unknown[]>
  transaction<T>(fn: (ctx: TransactionContext) => Promise<T>): Promise<T>
  getMetadata(key: string): Promise<unknown>
  setMetadata(key: string, value: unknown): Promise<void>
}
```
